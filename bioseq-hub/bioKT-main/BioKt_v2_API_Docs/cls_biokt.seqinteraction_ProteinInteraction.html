▶ Code Example
val seqA = ProteinSequence("MKALVLLYLLFSSAYSRGVFRRDTHKPP")
val seqB = ProteinSequence("ACDEFGHIKLMNPQRSTVWYACDEFGHI")

val ppi = ProteinInteraction.score(seqA, seqB)
println(ppi.confidence)          // 0.88
println(ppi.isLikelyInteracting) // true
println(ppi.evidence)
// [Charge complementarity, Hydrophobic patch match, High interface propensity]

val iface = ProteinInteraction.predictInterfaceResidues(seqA)
iface.filter { it.isLikelyInterface }
     .forEach { println("Pos ${it.position}: ${it.aminoAcid}") }

val cc = ProteinInteraction.predictCoiledCoil(seqA)
cc.forEach { println("Coiled coil ${it.start}–${it.end}, p=${it.probability}") }
