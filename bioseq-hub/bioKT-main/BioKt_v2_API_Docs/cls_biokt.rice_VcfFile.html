▶ Code Example
val vcf = VcfParser.parse("panel.vcf", snpsOnly=true, minMaf=0.05)

// Chain filters
val clean = vcf.biallelic().maxMissing(0.1).minMaf(0.01)

// Dosage matrix for ML / PCA
val X = clean.dosageMatrix()
println("Matrix: ${X.size} variants × ${X[0].size} samples")

// LD between two SNPs
val snp1 = clean.records[0]
val snp2 = clean.records[10]
val (r2, dprime) = clean.ldBetween(snp1, snp2)
println("r2=$r2  D'=$dprime")

// SNP density in 100kb windows
clean.snpDensity(100_000).take(5)
    .forEach { w -> println("${w.chrom}:${w.start}-${w.end}  ${w.value.toInt()} SNPs") }
