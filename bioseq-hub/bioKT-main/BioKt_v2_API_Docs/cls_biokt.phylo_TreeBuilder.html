▶ Code Example
val m = DistanceMatrix(
    labels = listOf("Human", "Chimp", "Mouse"),
    data = arrayOf(
        doubleArrayOf(0.0, 0.02, 0.25),
        doubleArrayOf(0.02, 0.0, 0.26),
        doubleArrayOf(0.25, 0.26, 0.0)
    )
)
val tree = TreeBuilder.neighborJoining(m)
println(tree.toNewick())          // ((Human:0.01,Chimp:0.01):0.12,Mouse:0.13);
println(tree.leaves().size)       // 3
