▶ Code Example
val aspirin = Molecule("CC(=O)Oc1ccccc1C(=O)O", name="Aspirin")
val d = MolDescriptors.calculate(aspirin)

println(d.molecularWeight)               // 180.16
println(d.logP)                          // 1.24
println(d.lipinskiPasses)               // true
println(d.admet.oralBioavailability)    // "High"
println(d.admet.bbbPenetration)         // true

val alerts = MolDescriptors.checkStructuralAlerts(aspirin)
// [] — no alerts for aspirin
