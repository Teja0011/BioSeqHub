▶ Code Example
val prot = ProteinSequence("MKALVLLYLLFSSAYSRGVFRRDTHKPPVYK")

// Single descriptors
println(ProteinDescriptors.gravyIndex(prot))       // −0.16
println(ProteinDescriptors.aliphaticIndex(prot))   // 94.9
println(ProteinDescriptors.chargeAtPH(prot, 7.0))  // +2.3

// Composition vectors
val aac = ProteinDescriptors.aminoAcidComposition(prot)
aac.entries.sortedByDescending { it.value }.take(3)
    .forEach { println("${it.key}=${it.value}") }
// L=0.154, K=0.103, P=0.103

// All descriptors for ML
val allDesc = ProteinDescriptors.calculateAll(prot)
val flatMap = allDesc.toFlatMap()  // ~500+ features
println(flatMap.size)
