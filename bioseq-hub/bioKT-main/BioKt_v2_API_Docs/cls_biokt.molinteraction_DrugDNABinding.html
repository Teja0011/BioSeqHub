▶ Code Example
val etBr = Molecule("CCn1c2cc(N)ccc2cc2ccc(N)cc12", name="Ethidium-like")
val dna  = DNASequence("ATCGATCGATCGATCG")

val p = DrugDNABinding.profile(etBr, dna)
println(p.bindingMode)           // "Minor groove binding"
println(p.intercalationScore)    // 3.25
println(p.deltaG)                // −2.84 kcal/mol
println(p.covalentRisk.riskLevel)// "Low"

// Covalent risk for an alkylating agent
val alkylator = Molecule("ClCC(=O)Cl")
val risk = DrugDNABinding.covalentBindingRisk(alkylator)
println(risk.riskLevel)          // "High"
println(risk.structuralAlerts)
